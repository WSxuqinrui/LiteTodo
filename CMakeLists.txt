cmake_minimum_required(VERSION 3.21)
project(LiteTodo VERSION 0.4.1 LANGUAGES C CXX)

# macOS 的平台适配和 FLTK 都含 Objective-C++，必须在依赖加入前于顶层启用。
# 否则子目录先启用语言后，顶层可能缺少 CMAKE_OBJCXX_COMPILE_OBJECT 规则。
if(APPLE)
  enable_language(OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build type")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# 只编译实际使用的 FLTK 核心，不把开发工具和额外图片库带入发布包。
set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_GL OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FORMS OFF CACHE BOOL "" FORCE)
set(FLTK_OPTION_SVG OFF CACHE BOOL "" FORCE)
set(FLTK_OPTION_PRINT_SUPPORT OFF CACHE BOOL "" FORCE)
set(FLTK_MSVC_RUNTIME_DLL OFF CACHE BOOL "" FORCE)
set(FLTK_USE_SYSTEM_ZLIB OFF CACHE BOOL "" FORCE)
set(FLTK_USE_SYSTEM_LIBPNG OFF CACHE BOOL "" FORCE)
set(FLTK_USE_SYSTEM_LIBJPEG OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
  set(FLTK_BACKEND_WAYLAND OFF CACHE BOOL "" FORCE)
  set(FLTK_USE_PANGO ON CACHE BOOL "" FORCE)
endif()

file(GLOB SQLITE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sqlite-amalgamation-*/sqlite3.c")
list(LENGTH SQLITE_SOURCES SQLITE_COUNT)
if(NOT SQLITE_COUNT EQUAL 1 OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fltk-release-1.4.5/CMakeLists.txt")
  message(FATAL_ERROR "Prepare pinned dependencies with scripts/fetch-deps.py")
endif()
list(GET SQLITE_SOURCES 0 SQLITE_SOURCE)
get_filename_component(SQLITE_INCLUDE "${SQLITE_SOURCE}" DIRECTORY)
add_library(sqlite_small STATIC "${SQLITE_SOURCE}")
target_include_directories(sqlite_small PUBLIC "${SQLITE_INCLUDE}")
# 保留线程安全、事务与备份接口；仅裁剪未使用的扩展。
target_compile_definitions(sqlite_small PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_DEPRECATED SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_DQS=0)
add_library(todo_core STATIC src/store.cpp src/paths.cpp src/dates.cpp src/workspace.cpp)
target_include_directories(todo_core PUBLIC src)
target_link_libraries(todo_core PUBLIC sqlite_small)

add_subdirectory(vendor/fltk-release-1.4.5 EXCLUDE_FROM_ALL)
if(WIN32)
  set(PLATFORM_SOURCE src/platform_windows.cpp)
elseif(APPLE)
  set(PLATFORM_SOURCE src/platform_macos.mm)
else()
  set(PLATFORM_SOURCE src/platform_linux.cpp)
endif()
add_executable(LiteTodo WIN32 MACOSX_BUNDLE src/main.cpp src/ui.cpp src/widgets.cpp src/icons.cpp src/theme.cpp src/settings.cpp ${PLATFORM_SOURCE})
target_link_libraries(LiteTodo PRIVATE todo_core fltk::fltk)
target_include_directories(LiteTodo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(WIN32)
  enable_language(RC)
  target_sources(LiteTodo PRIVATE src/app.rc)
  set_source_files_properties(src/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
  target_compile_definitions(LiteTodo PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(LiteTodo PRIVATE shell32 user32 gdi32 advapi32 ole32)
elseif(APPLE)
  target_sources(LiteTodo PRIVATE assets/app.icns)
  set_source_files_properties(assets/app.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_link_libraries(LiteTodo PRIVATE "-framework Cocoa" "-framework UserNotifications")
  set_source_files_properties(src/platform_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  set_target_properties(LiteTodo PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Info.plist.in")
else()
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DBUS IMPORTED_TARGET dbus-1)
  target_link_libraries(LiteTodo PRIVATE X11::X11)
  if(DBUS_FOUND)
    target_link_libraries(LiteTodo PRIVATE PkgConfig::DBUS)
    target_compile_definitions(LiteTodo PRIVATE LITETODO_DBUS)
  endif()
  target_link_libraries(sqlite_small PUBLIC ${CMAKE_DL_LIBS} pthread m)
endif()

enable_testing()
add_executable(store_tests tests/store_tests.cpp)
target_link_libraries(store_tests PRIVATE todo_core)
if(WIN32)
  target_link_libraries(store_tests PRIVATE shell32)
endif()
add_test(NAME store_contract COMMAND store_tests)

if(MSVC)
  target_compile_options(LiteTodo PRIVATE /utf-8 /W4)
  target_compile_options(todo_core PRIVATE /utf-8 /W4)
  target_compile_options(store_tests PRIVATE /utf-8)
  target_link_options(LiteTodo PRIVATE /OPT:REF /OPT:ICF)
else()
  foreach(target LiteTodo todo_core sqlite_small store_tests)
    target_compile_options(${target} PRIVATE -Os -ffunction-sections -fdata-sections)
  endforeach()
  if(MINGW)
    target_link_options(LiteTodo PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--gc-sections -s)
    target_link_options(store_tests PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(APPLE)
    target_link_options(LiteTodo PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(LiteTodo PRIVATE -Wl,--gc-sections -s)
  endif()
  target_compile_options(LiteTodo PRIVATE -Wall -Wextra)
  target_compile_options(todo_core PRIVATE -Wall -Wextra)
endif()
